<style>
/* Celle disabilitate della tabella */
.td-disabled {
    background-color: #f0f0f0;
    pointer-events: none;  /* blocca interazioni su input */
    opacity: 0.6;
}

/* Riabilita i pulsanti delete e modify anche se la cella è disabilitata */
.td-disabled button.DeleteRow,
.td-disabled button.ModifyRow {
    pointer-events: auto;
    opacity: 1;
}
</style>

 
 <script>
 let forceCounter = 1; // Contatore globale delle operazioni

function disableRow(row) {
    // Disabilita tutti i controlli tranne DeleteRow e ModifyRow
    row.find('input, select, textarea, button')
        .not('.DeleteRow, .ModifyRow')
        .prop('disabled', true);

    // Aggiunge classe di stile per la riga
    row.find('td').addClass('td-disabled');
}

function enableRow(row) {
    // Riabilita tutti i controlli tranne DeleteRow e ModifyRow
    row.find('input, select, textarea, button')
        .not('.DeleteRow, .ModifyRow')
        .prop('disabled', false);

    // Rimuove classe di stile
    row.find('td').removeClass('td-disabled');
}

// Gestione DeleteRow
function handleDeleteRow(button) {
    const row = $(button).closest('tr');
    const pc1Ienv = row.find('input[name$=".Pc1Ienv"]');

    // Aggiorna campo con FORCE[n]
    pc1Ienv.val("FORCE" + forceCounter++);
    
    // Rimuovi la riga dalla tabella
    row.remove();
}

// Gestione ModifyRow
function handleModifyRow(button) {
    const row = $(button).closest('tr');
    const pc1Ienv = row.find('input[name$=".Pc1Ienv"]');

    if (row.hasClass('editing')) {
        // Conferma modifica -> disabilita di nuovo la riga
        row.removeClass('editing');
        disableRow(row);
        pc1Ienv.val("FORCE" + forceCounter++);
    } else {
        // Abilita modifica
        row.addClass('editing');
        enableRow(row);
    }
}

// Gestione AddRow
function handleAddRow(button) {
    const table = $(button).closest('table');
    const newRow = table.find('tr.template-row').clone().removeClass('template-row');

    // Reset valori della riga
    newRow.find('input, select, textarea').val('');
    
    // Aggiorna campo con FORCE[n]
    const pc1Ienv = newRow.find('input[name$=".Pc1Ienv"]');
    pc1Ienv.val("FORCE" + forceCounter++);

    // Aggiunge la riga e disabilita di default
    table.append(newRow);
    disableRow(newRow);
}
</script>
